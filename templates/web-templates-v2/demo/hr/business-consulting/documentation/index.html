                            <!-- all plugin css -->
                            <link rel="stylesheet preload" href="assets/css/plugins.min.css" as="style">
                            <!-- fontawesome css -->
                            <link rel="stylesheet preload" href="assets/css/plugins/fontawesome.css" as="style">
                            <!-- odometer css -->
                            <link rel="stylesheet preload" href="assets/css/plugins/odometer.css" as="style">
                            <!-- main style custom css -->
                            <link rel="stylesheet preload" href="assets/css/style.css" as="style">

1.4. Source File (JS)

                            <!-- plugin js -->
                            <script src="assets/js/plugins.min.js"></script>
                            <!-- odometer js -->
                            <script src="assets/js/plugins/odometer.js"></script>
                            <!-- smooth scroll -->
                            <script src="assets/js/plugins/smooth-scroll.js"></script>
                            <!-- custom js -->
                            <script src="assets/js/main.js"></script>
